# Complete FastAPI Service with GitHub Webhooks & AST Scanner
from fastapi import FastAPI, Request, HTTPException
import hmac, hashlib, ast

app = FastAPI(title="CDX Autonomous Review MVP", version="1.7.0")

@app.post("/api/v1/webhook")
async def github_webhook(request: Request):
    payload = await request.body()
    signature = request.headers.get("X-Hub-Signature-256", "")
    # Constant-time comparison prevents timing attacks
    expected = "sha256=" + hmac.new(b"SECRET_KEY", payload, hashlib.sha256).hexdigest()
    if not hmac.compare_digest(signature, expected):
        raise HTTPException(status_code=403, detail="Invalid HMAC Signature")
    return {"status": "verified_and_queued", "model": "CDX 3.2 LPU Ultra"}
